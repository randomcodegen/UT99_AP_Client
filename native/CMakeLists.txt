cmake_minimum_required(VERSION 3.24)
project(UT99APNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/sdk" CACHE PATH "UT469e public SDK")
set(APCPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../APCpp" CACHE PATH "Local APCpp source")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_ROOT "${CMAKE_BINARY_DIR}/mbedtls-3.6.4")
if(NOT EXISTS "${MBEDTLS_ROOT}/CMakeLists.txt")
    file(ARCHIVE_EXTRACT INPUT "${APCPP}/mbedtls-3.6.4.tar.bz2" DESTINATION "${CMAKE_BINARY_DIR}")
endif()
add_subdirectory("${MBEDTLS_ROOT}" mbedtls)
set(USE_TLS ON CACHE BOOL "" FORCE)
set(USE_MBED_TLS ON CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(zlib SOURCE_DIR "${APCPP}/zlib" OVERRIDE_FIND_PACKAGE)
FetchContent_MakeAvailable(zlib)
add_library(ZLIB::ZLIB ALIAS zlibstatic)
set(USE_ZLIB ON CACHE BOOL "" FORCE)
set(IXWEBSOCKET_INSTALL OFF CACHE BOOL "" FORCE)
set(MBEDTLS_INCLUDE_DIRS "${MBEDTLS_ROOT}/include")
set(MBEDTLS_VERSION_GREATER_THAN_3 TRUE)
set(MBEDTLS_LIBRARY mbedtls)
set(MBEDX509_LIBRARY mbedx509)
set(MBEDCRYPTO_LIBRARY mbedcrypto)
add_subdirectory("${APCPP}/IXWebSocket" ixwebsocket)
target_compile_options(ixwebsocket PRIVATE /EHsc)
set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_POST_BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_PKGCONFIG_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${APCPP}/jsoncpp" jsoncpp)
target_compile_options(jsoncpp_static PRIVATE /EHsc)
if(TARGET jsoncpp_object)
    target_compile_options(jsoncpp_object PRIVATE /EHsc)
endif()
add_library(APSession STATIC Session.cpp "${APCPP}/Archipelago.cpp")
target_include_directories(APSession PRIVATE "${APCPP}" "${APCPP}/IXWebSocket" "${APCPP}/jsoncpp/include")
target_link_libraries(APSession PUBLIC ixwebsocket jsoncpp_static)
target_compile_options(APSession PRIVATE /EHsc)
add_library(UT99APNative SHARED Bridge.cpp)
target_include_directories(UT99APNative PRIVATE "${SDK}/Core/Inc" "${SDK}/Engine/Inc")
target_compile_definitions(UT99APNative PRIVATE UNICODE _UNICODE WIN32 ENGINE_ARCHITECTURE="x86" _CRT_SECURE_NO_WARNINGS _CRT_NO_VA_START_VALIDATION WINDOWS_IGNORE_PACKING_MISMATCH)
target_compile_options(UT99APNative PRIVATE /Zp4 /Zc:implicitNoexcept- /Zc:wchar_t- /EHa)
target_link_libraries(UT99APNative PRIVATE "${SDK}/Core/Lib/Core.lib" "${SDK}/Engine/Lib/Engine.lib" APSession)

option(UT99AP_BUILD_TESTS "Build native TLS integration probe" OFF)
if(UT99AP_BUILD_TESTS)
    add_executable(NativeSessionProbe ../tests/native_session_probe.cpp)
    target_include_directories(NativeSessionProbe PRIVATE . "${APCPP}/IXWebSocket" "${APCPP}/jsoncpp/include")
    target_link_libraries(NativeSessionProbe PRIVATE APSession)
    target_compile_options(NativeSessionProbe PRIVATE /EHsc)
endif()
